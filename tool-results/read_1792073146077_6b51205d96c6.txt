    82→import type { EmployeeDTO, EmployeeDetail, ScanEventDTO } from "@/lib/types";
    83→import { EmployeeDetailSheet } from "@/components/admin/views/employee-detail-sheet";
    84→import {
    85→  ViewContainer,
    86→  EmptyState,
    87→  TableSkeleton,

    99→} from "../lib/admin-utils";
   100→
   101→const PAGE_SIZE = 10;
   102→
   103→export function EmployeesView() {
   104→  const admin = useAppStore((s) => s.admin);
   105→  const setSection = useAppStore((s) => s.setAdminSection);
   106→  const setEnrollmentTargetId = useAppStore((s) => s.setEnrollmentTargetId);
   107→  const qc = useQueryClient();
   108→
   109→  const [page, setPage] = useState(1);
   110→  const [search, setSearch] = useState("");
   111→  const [deptFilter, setDeptFilter] = useState<string>("all");
   112→  const [statusFilter, setStatusFilter] = useState<string>("all");
   113→
   114→  const canEdit =
   115→    admin?.role === "hr_admin" || admin?.role === "superadmin";
   116→  const canRevoke =
   117→    admin?.role === "security_officer" || admin?.role === "superadmin";
   118→  const canDelete = canEdit;
   119→
   120→  const { data: departmentsRaw } = useQuery({
   121→    queryKey: ["departments"],
   122→    queryFn: async () =>
   123→      asArray<{ id: string; name: string; code: string }>(
   124→        await apiClient.departments.list(),
   125→      ),
   126→  });
   127→  const departments = departmentsRaw ?? [];
   128→
   129→  const { data, isLoading, isFetching } = useQuery({
   130→    queryKey: [
   131→      "employees",
   132→      page,
   133→      search,
   134→      deptFilter,
   135→      statusFilter,
   136→    ],
   137→    queryFn: () =>
   138→      apiClient.employees.list({
   139→        page,
   140→        q: search || undefined,
   141→        department_id: deptFilter === "all" ? undefined : deptFilter,
   142→        status: statusFilter === "all" ? undefined : statusFilter,
   143→      }),
   144→    placeholderData: (prev) => prev,
   145→  });
   146→
   147→  const totalPages = data ? Math.max(1, Math.ceil(data.total / data.pageSize)) : 1;
   148→  const employees = data?.items ?? [];
   149→
   150→  // -------- dialog state --------
   151→  const [createOpen, setCreateOpen] = useState(false);
   152→  const [editTarget, setEditTarget] = useState<EmployeeDTO | null>(null);
   153→  const [detailsTarget, setDetailsTarget] = useState<EmployeeDTO | null>(null);
   154→  const [detailsData, setDetailsData] = useState<EmployeeDetail | null>(null);
   155→  const [qrDialog, setQrDialog] = useState<{ token: string; employee: EmployeeDTO | EmployeeDetail } | null>(null);
   156→  const [revokeTarget, setRevokeTarget] = useState<EmployeeDTO | null>(null);
   157→  const [deleteTarget, setDeleteTarget] = useState<EmployeeDTO | null>(null);
   158→  const [historyTarget, setHistoryTarget] = useState<EmployeeDTO | null>(null);
   159→
   160→  // -------- mutations --------
   161→  const createMutation = useMutation({
   162→    mutationFn: (body: Record<string, unknown>) => apiClient.employees.create(body),
   163→    onSuccess: (res) => {
   164→      toast.success(`Employee ${res.employee.fullName} created`);
   165→      qc.invalidateQueries({ queryKey: ["employees"] });
   166→      setCreateOpen(false);
   167→      setQrDialog({ token: res.cardToken, employee: res.employee });
   168→    },
   169→    onError: (e: { error?: string }) => toast.error(e?.error ?? "Failed to create employee"),
   170→  });
   171→
   172→  const updateMutation = useMutation({
   173→    mutationFn: ({ id, body }: { id: string; body: Record<string, unknown> }) =>
   174→      apiClient.employees.update(id, body),
   175→    onSuccess: () => {
   176→      toast.success("Employee updated");
   177→      qc.invalidateQueries({ queryKey: ["employees"] });
   178→      setEditTarget(null);
   179→    },
   180→    onError: (e: { error?: string }) => toast.error(e?.error ?? "Update failed"),
   181→  });
   182→
   183→  const reissueMutation = useMutation({
   184→    mutationFn: (id: string) => apiClient.employees.reissueCard(id),
   185→    onSuccess: (res) => {
   186→      toast.success("New card issued — previous token revoked");
   187→      qc.invalidateQueries({ queryKey: ["employees"] });
   188→      setQrDialog({ token: res.cardToken, employee: res.employee });
   189→    },
   190→    onError: (e: { error?: string }) => toast.error(e?.error ?? "Reissue failed"),
   191→  });
   192→
   193→  const revokeMutation = useMutation({
   194→    mutationFn: ({
   195→      id,
   196→      body,
   197→    }: {
   198→      id: string;
   199→      body: { reason?: string; suspendFace?: boolean };
   200→    }) => apiClient.employees.revoke(id, body),
   201→    onSuccess: () => {
   202→      toast.success("Card revoked and CRL bumped");
   203→      qc.invalidateQueries({ queryKey: ["employees"] });
   204→      setRevokeTarget(null);
   205→    },
   206→    onError: (e: { error?: string }) => toast.error(e?.error ?? "Revoke failed"),
   207→  });
   208→
   209→  const deleteMutation = useMutation({
   210→    mutationFn: (id: string) => apiClient.employees.remove(id),
   211→    onSuccess: () => {
   212→      toast.success("Employee deleted (soft-delete)");
   213→      qc.invalidateQueries({ queryKey: ["employees"] });
   214→      setDeleteTarget(null);
   215→    },
   216→    onError: (e: { error?: string }) => toast.error(e?.error ?? "Delete failed"),
   217→  });
   218→
   219→  function startEnrollment(emp: EmployeeDTO) {
   220→    setEnrollmentTargetId(emp.id);
   221→    setSection("enrollment");
   222→  }
   223→
   224→  return (
   225→    <ViewContainer>
   226→      {/* Header */}
   227→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   228→        <div>
   229→          <h2 className="text-xl font-semibold flex items-center gap-2">
   230→            <Users className="size-5 text-primary" />
   231→            Employees
   232→          </h2>
   233→          <p className="text-sm text-muted-foreground mt-0.5">
   234→            {data?.total ?? 0} total · manage cards, enrollment, and access levels
   235→          </p>
   236→        </div>
   237→        {canEdit && (
   238→          <Button onClick={() => setCreateOpen(true)}>
   239→            <Plus className="size-4" />
   240→            Add Employee
   241→          </Button>
   242→        )}
   243→      </div>
   244→
   245→      {/* Filters */}
   246→      <Card className="p-4">
   247→        <div className="flex flex-col md:flex-row gap-3">
   248→          <div className="relative flex-1">
   249→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   250→            <Input
   251→              placeholder="Search by name, code, or designation…"
   252→              value={search}
   253→              onChange={(e) => {
   254→                setSearch(e.target.value);
   255→                setPage(1);
   256→              }}
   257→              className="pl-9"
   258→            />
   259→          </div>
   260→          <Select
   261→            value={deptFilter}
   262→            onValueChange={(v) => {
   263→              setDeptFilter(v);
   264→              setPage(1);
   265→            }}
   266→          >
   267→            <SelectTrigger className="w-full md:w-52">
   268→              <Filter className="size-4 mr-1.5 text-muted-foreground" />
   269→              <SelectValue placeholder="Department" />
   270→            </SelectTrigger>
   271→            <SelectContent>
   272→              <SelectItem value="all">All departments</SelectItem>
   273→              {departments?.map((d) => (
   274→                <SelectItem key={d.id} value={d.id}>
   275→                  {d.name}
   276→                </SelectItem>
   277→              ))}
   278→            </SelectContent>
   279→          </Select>
   280→          <Select
   281→            value={statusFilter}
   282→            onValueChange={(v) => {
   283→              setStatusFilter(v);
   284→              setPage(1);
   285→            }}
   286→          >
   287→            <SelectTrigger className="w-full md:w-44">
   288→              <SelectValue placeholder="Status" />
   289→            </SelectTrigger>
   290→            <SelectContent>
   291→              <SelectItem value="all">All statuses</SelectItem>
   292→              <SelectItem value="active">Active</SelectItem>
   293→              <SelectItem value="revoked">Revoked</SelectItem>
   294→              <SelectItem value="expired">Expired</SelectItem>
   295→              <SelectItem value="lost">Lost</SelectItem>
   296→            </SelectContent>
   297→          </Select>
   298→        </div>
   299→      </Card>
   300→
   301→      {/* Table */}
   302→      <Card className="p-0 overflow-hidden">
   303→        {isLoading ? (
   304→          <div className="p-4">
   305→            <TableSkeleton rows={6} cols={6} />
   306→          </div>
   307→        ) : employees.length === 0 ? (
   308→          <EmptyState
   309→            icon={Users}
   310→            title="No employees found"
   311→            description="Try adjusting your search or filters, or create a new employee."
   312→            action={
   313→              canEdit && (
   314→                <Button onClick={() => setCreateOpen(true)}>
   315→                  <Plus className="size-4" />
   316→                  Add Employee
   317→                </Button>
   318→              )
   319→            }
   320→          />
   321→        ) : (
   322→          <div className="overflow-x-auto scrollbar-thin">
   323→            <table className="w-full text-sm">
   324→              <thead>
   325→                <tr className="border-b border-border text-xs uppercase tracking-wider text-muted-foreground bg-muted/40">
   326→                  <th className="text-left font-medium py-3 px-4">Employee</th>
   327→                  <th className="text-left font-medium py-3 px-4 hidden md:table-cell">Department</th>
   328→                  <th className="text-left font-medium py-3 px-4 hidden lg:table-cell">Designation</th>
   329→                  <th className="text-left font-medium py-3 px-4">Access</th>
   330→                  <th className="text-left font-medium py-3 px-4">Card Status</th>
   331→                  <th className="text-left font-medium py-3 px-4 hidden md:table-cell">Face</th>
   332→                  <th className="text-left font-medium py-3 px-4 hidden lg:table-cell">Card Expires</th>
   333→                  <th className="text-right font-medium py-3 px-4 w-12"></th>
   334→                </tr>
   335→              </thead>
   336→              <tbody>
   337→                {employees.map((emp) => (
   338→                  <motion.tr
   339→                    key={emp.id}
   340→                    initial={{ opacity: 0 }}
   341→                    animate={{ opacity: 1 }}
   342→                    transition={{ duration: 0.2 }}
   343→                    className="border-b border-border/40 hover:bg-accent/30 transition-colors group"
   344→                  >
   345→                    <td className="py-3 px-4">
   346→                      <button
   347→                        onClick={() => openDetails(emp)}
   348→                        className="flex items-center gap-3 text-left"
   349→                      >
   350→                        <Avatar className="size-9 border border-border">
   351→                          {emp.photoUrl ? (
   352→                            <AvatarImage src={emp.photoUrl} alt={emp.fullName} />
   353→                          ) : null}
   354→                          <AvatarFallback className="bg-primary/15 text-primary text-xs">
   355→                            {initials(emp.fullName)}
   356→                          </AvatarFallback>
   357→                        </Avatar>
   358→                        <div className="min-w-0">
   359→                          <p className="font-medium text-sm truncate">{emp.fullName}</p>
   360→                          <p className="text-[11px] text-muted-foreground font-mono truncate">
   361→                            {emp.employeeCode}
   362→                          </p>
   363→                        </div>
   364→                      </button>
   365→                    </td>
   366→                    <td className="py-3 px-4 text-xs hidden md:table-cell">
   367→                      {emp.departmentName ?? "—"}
   368→                    </td>
   369→                    <td className="py-3 px-4 text-xs text-muted-foreground hidden lg:table-cell">
   370→                      {emp.designation ?? "—"}
   371→                    </td>
   372→                    <td className="py-3 px-4">
   373→                      <Badge
   374→                        variant="outline"
   375→                        className={cn("text-[10px]", accessLevelClass(emp.accessLevel))}
   376→                      >
   377→                        L{emp.accessLevel}
   378→                      </Badge>
   379→                    </td>
   380→                    <td className="py-3 px-4">
   381→                      <Badge
   382→                        variant="outline"
   383→                        className={cn("text-[10px]", cardStatusClass[emp.cardStatus])}
   384→                      >
   385→                        {emp.cardStatus}
   386→                      </Badge>
   387→                    </td>
   388→                    <td className="py-3 px-4 hidden md:table-cell">
   389→                      {emp.faceEnrolled ? (
   390→                        <span className="inline-flex items-center gap-1 text-xs text-success">
   391→                          <Fingerprint className="size-3.5" />
   392→                          Enrolled
   393→                        </span>
   394→                      ) : (
   395→                        <span className="inline-flex items-center gap-1 text-xs text-muted-foreground">
   396→                          <Fingerprint className="size-3.5 opacity-50" />
   397→                          None
   398→                        </span>
   399→                      )}
   400→                    </td>
   401→                    <td className="py-3 px-4 text-xs text-muted-foreground hidden lg:table-cell tabular-nums">
   402→                      {emp.cardExpiresAt
   403→                        ? formatRelative(emp.cardExpiresAt)
   404→                        : "—"}
   405→                    </td>
   406→                    <td className="py-3 px-4 text-right">
   407→                      <DropdownMenu>
   408→                        <DropdownMenuTrigger asChild>
   409→                          <Button
   410→                            variant="ghost"
   411→                            size="icon"
   412→                            className="size-8 opacity-60 group-hover:opacity-100"
   413→                          >
   414→                            <MoreHorizontal className="size-4" />
   415→                          </Button>
   416→                        </DropdownMenuTrigger>
   417→                        <DropdownMenuContent align="end" className="w-52">
   418→                          <DropdownMenuLabel>Actions</DropdownMenuLabel>
   419→                          <DropdownMenuItem
   420→                            onClick={() => openDetails(emp)}
   421→                            className="cursor-pointer"
   422→                          >
   423→                            <Eye className="size-4" />
   424→                            View Details
   425→                          </DropdownMenuItem>
   426→                          {canEdit && (
   427→                            <>
   428→                              <DropdownMenuItem
   429→                                onClick={() => setEditTarget(emp)}
   430→                                className="cursor-pointer"
   431→                              >
   432→                                <CreditCard className="size-4" />
   433→                                Edit Profile
   434→                              </DropdownMenuItem>
   435→                              <DropdownMenuItem
   436→                                onClick={() => reissueMutation.mutate(emp.id)}
   437→                                className="cursor-pointer"
   438→                                disabled={reissueMutation.isPending}
   439→                              >
   440→                                <CreditCard className="size-4" />
   441→                                Reissue Card
   442→                              </DropdownMenuItem>
   443→                            </>
   444→                          )}
   445→                          {(canEdit || admin?.role === "hr_admin") && (
   446→                            <DropdownMenuItem
   447→                              onClick={() => startEnrollment(emp)}
   448→                              className="cursor-pointer"
   449→                            >
   450→                              <ScanFace className="size-4" />
   451→                              Enroll Face
   452→                            </DropdownMenuItem>
   453→                          )}
   454→                          {canRevoke && (
   455→                            <DropdownMenuItem
   456→                              onClick={() => setRevokeTarget(emp)}
   457→                              className="cursor-pointer text-destructive focus:text-destructive"
   458→                            >
   459→                              <Ban className="size-4" />
   460→                              Revoke Card
   461→                            </DropdownMenuItem>
   462→                          )}
   463→                          <DropdownMenuItem
   464→                            onClick={() => setHistoryTarget(emp)}
   465→                            className="cursor-pointer"
   466→                          >
   467→                            <History className="size-4" />
   468→                            Scan History
   469→                          </DropdownMenuItem>
   470→                          {canDelete && (
   471→                            <>
   472→                              <DropdownMenuSeparator />
   473→                              <DropdownMenuItem
   474→                                onClick={() => setDeleteTarget(emp)}
   475→                                className="cursor-pointer text-destructive focus:text-destructive"
   476→                              >
   477→                                <Trash2 className="size-4" />
   478→                                Delete
   479→                              </DropdownMenuItem>
   480→                            </>
   481→                          )}
   482→                        </DropdownMenuContent>
   483→                      </DropdownMenu>
   484→                    </td>
   485→                  </motion.tr>
   486→                ))}
   487→              </tbody>
   488→            </table>
   489→          </div>
   490→        )}
   491→      </Card>
   492→
   493→      {/* Pagination */}
   494→      {data && data.total > data.pageSize && (
   495→        <div className="flex items-center justify-between gap-2">
   496→          <p className="text-xs text-muted-foreground">
   497→            Page {page} of {totalPages} · {data.total} total
   498→          </p>
   499→          <div className="flex gap-1.5">
   500→            <Button
   501→              variant="outline"
   502→              size="sm"
   503→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   504→              disabled={page === 1 || isFetching}
   505→            >
   506→              <ChevronLeft className="size-4" />
   507→              Prev
   508→            </Button>
   509→            <Button
   510→              variant="outline"
   511→              size="sm"
   512→              onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   513→              disabled={page >= totalPages || isFetching}
   514→            >
   515→              Next
   516→              <ChevronRight className="size-4" />
   517→            </Button>
   518→          </div>
   519→        </div>
   520→      )}
   521→
   522→      {/* Dialogs */}
   523→      {createOpen && (
   524→        <CreateEditDialog
   525→          key="new-employee"
   526→          open={createOpen}
   527→          onOpenChange={setCreateOpen}
   528→          departments={departments ?? []}
   529→          onSubmit={(b) => createMutation.mutate(b)}
   530→          loading={createMutation.isPending}
   531→        />
   532→      )}
   533→
   534→      {editTarget && (
   535→        <CreateEditDialog
   536→          key={`edit-${editTarget.id}`}
   537→          open={!!editTarget}
   538→          onOpenChange={(o) => !o && setEditTarget(null)}
   539→          departments={departments ?? []}
   540→          employee={editTarget}
   541→          onSubmit={(b) =>
   542→            updateMutation.mutate({ id: editTarget.id, body: b })
   543→          }
   544→          loading={updateMutation.isPending}
   545→        />
   546→      )}
   547→
   548→      <DetailsDialog
   549→        target={detailsTarget}
   550→        data={detailsData}
   551→        setData={setDetailsData}
   552→        onClose={() => {
   553→          setDetailsTarget(null);
   554→          setDetailsData(null);
   555→        }}
   556→      />
   557→
   558→      <EmployeeDetailSheet
   559→        employee={detailsTarget}
   560→        detail={detailsData}
   561→        open={!!detailsTarget}
   562→        onOpenChange={(o) => {
   563→          if (!o) {
   564→            setDetailsTarget(null);
   565→            setDetailsData(null);
   566→          }
   567→        }}
   568→      />
   569→
   570→      <QrDialog
   571→        token={qrDialog?.token ?? null}
   572→        employee={qrDialog?.employee ?? null}
   573→        onClose={() => setQrDialog(null)}
   574→      />
   575→
   576→      {revokeTarget && (
   577→        <RevokeDialog
   578→          employee={revokeTarget}
   579→          loading={revokeMutation.isPending}
   580→          onClose={() => setRevokeTarget(null)}
   581→          onConfirm={(body) =>
   582→            revokeMutation.mutate({ id: revokeTarget.id, body })
   583→          }
   584→        />
   585→      )}
   586→
   587→      {deleteTarget && (
   588→        <AlertDialog open onOpenChange={(o) => !o && setDeleteTarget(null)}>
   589→          <AlertDialogContent>
   590→            <AlertDialogHeader>
   591→              <AlertDialogTitle className="flex items-center gap-2">
   592→                <Trash2 className="size-5 text-destructive" />
   593→                Delete {deleteTarget.fullName}?
   594→              </AlertDialogTitle>
   595→              <AlertDialogDescription>
   596→                This performs a soft-delete. The employee record will be anonymized
   597→                to &ldquo;Deleted User&rdquo;, face embedding cleared, card token
   598→                revoked, and access permanently disabled. Audit trail entries are
   599→                retained immutably.
   600→              </AlertDialogDescription>
   601→            </AlertDialogHeader>
